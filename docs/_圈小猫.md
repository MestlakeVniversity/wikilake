\<script\> /\*\*

`* 游戏核心逻辑`\
`* 使用六边形网格坐标系 (Axial Coordinates)`\
`*/`

class HexGame {

`   constructor(canvasId) {`\
`       this.canvas = document.getElementById(canvasId);`\
`       this.ctx = this.canvas.getContext('2d');`\
`       this.width = this.canvas.width;`\
`       this.height = this.canvas.height;`\
`       `\
`       // 游戏配置`\
`       this.gridSize = 9; // 网格半径，越大地图越大`\
`       this.hexRadius = 30; // 六边形半径`\
`       this.hexHeight = Math.sqrt(3) * this.hexRadius;`\
`       `\
`       // 状态`\
`       this.grid = {}; // 存储格子状态：0=空，1=墙，2=猫`\
`       this.catPos = { q: 0, r: 0 };`\
`       this.steps = 0;`\
`       this.isGameOver = false;`\
`       this.isAnimating = false;`

`       // 初始化`\
`       this.initInput();`\
`       this.reset();`\
`       this.draw();`\
`   }`

`   // 重置游戏`\
`   reset() {`\
`       this.grid = {};`\
`       this.steps = 0;`\
`       this.isGameOver = false;`\
`       this.isAnimating = false;`\
`       this.catPos = { q: 0, r: 0 };`\
`       document.getElementById('message').style.display = 'none';`\
`       document.getElementById('step-count').innerText = '0';`\
`       `\
`       // 随机生成初始墙壁（边缘）`\
`       this.generateRandomWalls();`\
`       `\
`       this.draw();`\
`   }`

`   // 生成随机墙壁`\
`   generateRandomWalls() {`\
`       // 简单的随机策略：在边缘随机放一些墙`\
`       // 为了增加趣味性，我们只在最外层随机放`\
`       for (let q = -this.gridSize; q <= this.gridSize; q++) {`\
`           let r1 = Math.max(-this.gridSize, -q - this.gridSize);`\
`           let r2 = Math.min(this.gridSize, -q + this.gridSize);`\
`           for (let r = r1; r <= r2; r++) {`\
`               // 排除中心区域，让猫有活动空间`\
`               if (Math.abs(q) + Math.abs(r) + Math.abs(-q-r) > this.gridSize * 1.5) {`\
`                   if (Math.random() < 0.3) { // 30% 概率生成墙`\
`                       this.setCell(q, r, 1);`\
`                   }`\
`               }`\
`           }`\
`       }`\
`   }`

`   // 六边形坐标转屏幕坐标`\
`   hexToPixel(q, r) {`\
`       const x = this.width / 2 + this.hexRadius * (3/2 * q);`\
`       const y = this.height / 2 + this.hexRadius * (Math.sqrt(3)/2 * q + Math.sqrt(3) * r);`\
`       return { x, y };`\
`   }`

`   // 屏幕坐标转六边形坐标 (近似)`\
`   pixelToHex(x, y) {`\
`       const cx = this.width / 2;`\
`       const cy = this.height / 2;`\
`       `\
`       const q = (2/3 * x - 1/3 * y) / this.hexRadius; // 这里的公式是基于尖顶六边形的变体，需适配平顶`\
`       // 修正为平顶六边形公式：`\
`       // x = size * 3/2 * q`\
`       // y = size * sqrt(3) * (r + q/2)`\
`       `\
`       const q_float = (2/3 * x) / this.hexRadius;`\
`       const r_float = (-1/3 * x + Math.sqrt(3)/3 * y) / this.hexRadius;`\
`       `\
`       // 这里的转换逻辑比较复杂，简化处理：遍历查找最近的中心点`\
`       // 或者使用更精确的算法。为了代码简洁，我们使用遍历法检测点击区域。`\
`       // 实际上，对于这种小网格，遍历所有格子检测距离是最稳健的。`\
`       `\
`       let bestDist = Infinity;`\
`       let bestQ = 0, bestR = 0;`

`       for (let qq = -this.gridSize; qq <= this.gridSize; qq++) {`\
`           let r1 = Math.max(-this.gridSize, -qq - this.gridSize);`\
`           let r2 = Math.min(this.gridSize, -qq + this.gridSize);`\
`           for (let rr = r1; rr <= r2; rr++) {`\
`               const pos = this.hexToPixel(qq, rr);`\
`               const dist = Math.hypot(pos.x - x, pos.y - y);`\
`               if (dist < bestDist) {`\
`                   bestDist = dist;`\
`                   bestQ = qq;`\
`                   bestR = rr;`\
`               }`\
`           }`\
`       }`\
`       `\
`       // 如果点击距离中心太远，视为无效`\
`       if (bestDist > this.hexRadius) return null;`\
`       return { q: bestQ, r: bestR };`\
`   }`

`   setCell(q, r, type) {`\
``        const key = `${q},${r}`; ``\
`       this.grid[key] = type;`\
`   }`

`   getCell(q, r) {`\
``        return this.grid[`${q},${r}`] || 0; ``\
`   }`

`   // 获取六个邻居`\
`   getNeighbors(q, r) {`\
`       const directions = [`\
`           {q: 1, r: 0}, {q: 1, r: -1}, {q: 0, r: -1},`\
`           {q: -1, r: 0}, {q: -1, r: 1}, {q: 0, r: 1}`\
`       ];`\
`       return directions.map(d => ({ q: q + d.q, r: r + d.r }));`\
`   }`

`   // 核心 AI：BFS 寻路`\
`   // 返回猫应该移动到的下一个位置，如果无路可走返回 null`\
`   findBestMove() {`\
`       // 如果猫已经在边界，它赢了`\
`       if (Math.abs(this.catPos.q) >= this.gridSize || `\
`           Math.abs(this.catPos.r) >= this.gridSize || `\
`           Math.abs(-this.catPos.q - this.catPos.r) >= this.gridSize) {`\
`           return 'ESCAPE';`\
`       }`

`       // BFS 寻找离边界最近的路径`\
`       let queue = [{ q: this.catPos.q, r: this.catPos.r, dist: 0 }];`\
`       let visited = new Set();`\
``        visited.add(`${this.catPos.q},${this.catPos.r}`); ``

`       let bestNextMove = null;`\
`       let minDistToEdge = Infinity;`

`       // 简单的贪婪算法：每一步都走向离边界最近的格子`\
`       // 真正的 BFS 是找最短路径到边界，这里为了简化，我们模拟猫每一步都想往外跑`\
`       `\
`       // 获取所有可能的下一步`\
`       const neighbors = this.getNeighbors(this.catPos.q, this.catPos.r);`\
`       `\
`       // 过滤掉墙和已经访问过的（防止原地打转，虽然 BFS 会处理）`\
`       let candidates = neighbors.filter(n => {`\
`           // 检查是否在地图内`\
`           if (Math.abs(n.q) > this.gridSize || Math.abs(n.r) > this.gridSize || Math.abs(-n.q-n.r) > this.gridSize) return false;`\
`           // 检查是否是墙`\
`           if (this.getCell(n.q, n.r) === 1) return false;`\
`           return true;`\
`       });`

`       if (candidates.length === 0) return null; // 被围住了`

`       // 评估哪个邻居离边界“最近” (曼哈顿距离的变种)`\
`       // 边界条件：|q| >= size OR |r| >= size OR |s| >= size (s = -q-r)`\
`       // 距离边界的距离 = size - max(|q|, |r|, |s|)`\
`       `\
`       let bestScore = -Infinity;`\
`       `\
`       candidates.forEach(n => {`\
`           const s = -n.q - n.r;`\
`           const distToEdge = this.gridSize - Math.max(Math.abs(n.q), Math.abs(n.r), Math.abs(s));`\
`           `\
`           // 分数越高越好（距离边界越近，分数越小，我们要找最小的 distToEdge）`\
`           // 我们想要 distToEdge 最小`\
`           if (distToEdge < bestScore) {`\
`               bestScore = distToEdge;`\
`               bestNextMove = n;`\
`           } else if (distToEdge === bestScore) {`\
`               // 如果距离一样，随机选一个，增加不可预测性`\
`               if (Math.random() > 0.5) bestNextMove = n;`\
`           }`\
`       });`

`       return bestNextMove;`\
`   }`

`   initInput() {`\
`       this.canvas.addEventListener('click', (e) => {`\
`           if (this.isGameOver || this.isAnimating) return;`

`           const rect = this.canvas.getBoundingClientRect();`\
`           const x = e.clientX - rect.left;`\
`           const y = e.clientY - rect.top;`\
`           `\
`           const hex = this.pixelToHex(x, y);`\
`           `\
`           if (hex) {`\
`               // 只能点击空格子`\
`               if (this.getCell(hex.q, hex.r) === 0) {`\
`                   this.setCell(hex.q, hex.r, 1); // 放置墙`\
`                   this.steps++;`\
`                   document.getElementById('step-count').innerText = this.steps;`\
`                   this.draw();`\
`                   `\
`                   // 检查是否围住`\
`                   if (!this.checkWin()) {`\
`                       this.moveCat();`\
`                   }`\
`               }`\
`           }`\
`       });`\
`   }`

`   checkWin() {`\
`       const nextMove = this.findBestMove();`\
`       if (nextMove === null) {`\
`           this.endGame(true);`\
`           return true;`\
`       }`\
`       return false;`\
`   }`

`   moveCat() {`\
`       this.isAnimating = true;`\
`       const nextPos = this.findBestMove();`\
`       `\
`       if (nextPos === 'ESCAPE') {`\
`           this.endGame(false);`\
`           return;`\
`       }`

`       // 简单的动画延迟`\
`       setTimeout(() => {`\
`           this.catPos = nextPos;`\
`           this.isAnimating = false;`\
`           this.draw();`\
`           `\
`           // 移动后再次检查（防止瞬移出去）`\
`           if (Math.abs(this.catPos.q) >= this.gridSize || `\
`               Math.abs(this.catPos.r) >= this.gridSize || `\
`               Math.abs(-this.catPos.q - this.catPos.r) >= this.gridSize) {`\
`               this.endGame(false);`\
`           }`\
`       }, 300);`\
`   }`

`   endGame(win) {`\
`       this.isGameOver = true;`\
`       const msgBox = document.getElementById('message');`\
`       const title = document.getElementById('msg-title');`\
`       const content = document.getElementById('msg-content');`\
`       `\
`       msgBox.style.display = 'block';`\
`       if (win) {`\
`           title.innerText = "🎉 恭喜！你抓住了小猫！";`\
`           title.style.color = "#4CAF50";`\
``            content.innerText = `你用了 ${this.steps} 步。`; ``\
`       } else {`\
`           title.innerText = "😿 小猫逃跑了！";`\
`           title.style.color = "#f44336";`\
`           content.innerText = "别灰心，再试一次吧！";`\
`       }`\
`   }`

`   drawHex(q, r, color, strokeColor = '#ccc') {`\
`       const center = this.hexToPixel(q, r);`\
`       this.ctx.beginPath();`\
`       for (let i = 0; i < 6; i++) {`\
`           const angle = 2 * Math.PI / 6 * i;`\
`           const x_i = center.x + this.hexRadius * Math.cos(angle);`\
`           const y_i = center.y + this.hexRadius * Math.sin(angle);`\
`           if (i === 0) this.ctx.moveTo(x_i, y_i);`\
`           else this.ctx.lineTo(x_i, y_i);`\
`       }`\
`       this.ctx.closePath();`\
`       this.ctx.fillStyle = color;`\
`       this.ctx.fill();`\
`       this.ctx.strokeStyle = strokeColor;`\
`       this.ctx.lineWidth = 1;`\
`       this.ctx.stroke();`\
`   }`

`   draw() {`\
`       this.ctx.clearRect(0, 0, this.width, this.height);`\
`       `\
`       // 绘制背景网格`\
`       for (let q = -this.gridSize; q <= this.gridSize; q++) {`\
`           let r1 = Math.max(-this.gridSize, -q - this.gridSize);`\
`           let r2 = Math.min(this.gridSize, -q + this.gridSize);`\
`           for (let r = r1; r <= r2; r++) {`\
`               this.drawHex(q, r, '#eef2f5');`\
`           }`\
`       }`

`       // 绘制已放置的墙`\
`       for (let key in this.grid) {`\
`           if (this.grid[key] === 1) {`\
`               const [q, r] = key.split(',').map(Number);`\
`               this.drawHex(q, r, '#3498db', '#2980b9'); // 蓝色墙`\
`           }`\
`       }`

`       // 绘制小猫`\
`       const catCenter = this.hexToPixel(this.catPos.q, this.catPos.r);`\
`       this.drawHex(this.catPos.q, this.catPos.r, '#fff'); // 先画个白底`\
`       `\
`       // 画猫脸 (简单的 emoji 或图形)`\
``        this.ctx.font = `${this.hexRadius * 1.2}px Arial`; ``\
`       this.ctx.textAlign = 'center';`\
`       this.ctx.textBaseline = 'middle';`\
`       this.ctx.fillText('🐱', catCenter.x, catCenter.y + 2);`\
`   }`

}
